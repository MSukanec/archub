DECLARE
  -- ============================================================
  -- 🧑‍💻 Datos provenientes del auth.users (NEW)
  -- ============================================================
  new_user_id uuid := gen_random_uuid();
  auth_user_id uuid := NEW.id;
  user_full_name text := COALESCE(NEW.raw_user_meta_data->>'full_name', NULL);
  user_email text := LOWER(NEW.email);
  user_avatar text := COALESCE(NEW.raw_user_meta_data->>'avatar_url', '');
  provider text := LOWER(COALESCE(NEW.raw_user_meta_data->>'provider', 'email'));

  avatar_src public.avatar_source_t := 'email'::public.avatar_source_t;

  -- ============================================================
  -- 🎛 Valores globales
  -- ============================================================
  plan_free_id uuid := '015d8a97-6b6e-4aec-87df-5d1e6b0e4ed2';
  archub_user_role_id uuid := 'e6cc68d2-fc28-421b-8bd3-303326ef91b8';

  default_currency_id uuid := '58c50aa7-b8b1-4035-b509-58028dd0e33f';
  default_wallet_id uuid := '2658c575-0fa8-4cf6-85d7-6430ded7e188';
  default_pdf_template_id uuid := 'b6266a04-9b03-4f3a-af2d-f6ee6d0a948b';

  -- ============================================================
  -- 🏢 Organización
  -- ============================================================
  org_id uuid := gen_random_uuid();
  org_name text := 'Organización de ' || COALESCE(user_full_name, 'Usuario');

  -- ============================================================
  -- 👥 Roles de la organización
  -- ============================================================
  admin_role_id uuid := gen_random_uuid();
  editor_role_id uuid := gen_random_uuid();
  viewer_role_id uuid := gen_random_uuid();

  -- ============================================================
  -- ⏱️ Tiempo
  -- ============================================================
  now_timestamp timestamptz := NOW();

BEGIN
  -- ============================================================
  -- 🖼 Avatar source
  -- ============================================================
  IF provider = 'google' THEN
    avatar_src := 'google';
  ELSIF provider = 'discord' THEN
    avatar_src := 'discord';
  ELSE
    avatar_src := 'email';
  END IF;

  -- ============================================================
  -- 1) users
  -- ============================================================
  INSERT INTO public.users (
    id,
    auth_id,
    email,
    full_name,
    avatar_url,
    avatar_source,
    role_id
  )
  VALUES (
    new_user_id,
    auth_user_id,
    user_email,
    user_full_name,
    user_avatar,
    avatar_src,
    archub_user_role_id
  );

  -- ============================================================
  -- 2) user_data
  -- ============================================================
  INSERT INTO public.user_data (
    id,
    user_id,
    created_at
  )
  VALUES (
    gen_random_uuid(),
    new_user_id,
    now_timestamp
  );

  -- ============================================================
  -- 3) organizations
  -- ============================================================
  INSERT INTO public.organizations (
    id,
    name,
    created_by,
    owner_id,
    created_at,
    updated_at,
    is_active,
    plan_id
  )
  VALUES (
    org_id,
    org_name,
    new_user_id,
    new_user_id,
    now_timestamp,
    now_timestamp,
    TRUE,
    plan_free_id
  );

  -- ============================================================
  -- 4) organization_data
  -- ============================================================
  INSERT INTO public.organization_data (
    organization_id
  )
  VALUES (
    org_id
  );

-- ============================================================
-- 4.5) ROLES BASE DE LA ORGANIZACIÓN (ESQUEMA REAL)
-- ============================================================
INSERT INTO public.roles (
  id, name, description, type, organization_id, is_system
)
VALUES
  (admin_role_id,  'Administrador', 'Acceso total',        'organization', org_id, FALSE),
  (editor_role_id, 'Editor',        'Puede editar',        'organization', org_id, FALSE),
  (viewer_role_id, 'Lector',        'Solo lectura',        'organization', org_id, FALSE);


  -- ============================================================
  -- 5) organization_members (creator = admin)
  -- ============================================================
  INSERT INTO public.organization_members (
    id,
    user_id,
    organization_id,
    role_id,
    is_active,
    created_at,
    joined_at,
    last_active_at
  )
  VALUES (
    gen_random_uuid(),
    new_user_id,
    org_id,
    admin_role_id,
    TRUE,
    now_timestamp,
    now_timestamp,
    now_timestamp
  );

  -- ============================================================
  -- 6) organization_preferences
  -- ============================================================
  INSERT INTO public.organization_preferences (
    organization_id,
    default_currency_id,
    default_wallet_id,
    default_pdf_template_id,
    use_currency_exchange,
    created_at,
    updated_at
  )
  VALUES (
    org_id,
    default_currency_id,
    default_wallet_id,
    default_pdf_template_id,
    FALSE,
    now_timestamp,
    now_timestamp
  );

  -- ============================================================
  -- 7) user_preferences
  -- ============================================================
  INSERT INTO public.user_preferences (
    id,
    user_id,
    last_organization_id,
    theme,
    onboarding_completed,
    sidebar_docked,
    layout,
    created_at,
    updated_at
  )
  VALUES (
    gen_random_uuid(),
    new_user_id,
    org_id,
    'light',
    FALSE,
    FALSE,
    'experimental',
    now_timestamp,
    now_timestamp
  );

  -- ============================================================
  -- 8) user_organization_preferences
  -- ============================================================
  INSERT INTO public.user_organization_preferences (
    id,
    user_id,
    organization_id,
    created_at,
    updated_at
  )
  VALUES (
    gen_random_uuid(),
    new_user_id,
    org_id,
    now_timestamp,
    now_timestamp
  );

  RETURN NEW;

EXCEPTION
  WHEN OTHERS THEN
    RAISE EXCEPTION '💥 ERROR en handle_new_user: %', SQLERRM;
END;

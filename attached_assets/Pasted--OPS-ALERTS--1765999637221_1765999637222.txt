-- =========================================================
-- OPS: ALERTS
-- =========================================================
create table if not exists public.ops_alerts (
  id uuid primary key default gen_random_uuid(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  severity text not null default 'high',  -- 'low'|'medium'|'high'|'critical'
  status text not null default 'open',    -- 'open'|'ack'|'resolved'|'dismissed'

  alert_type text not null,              -- e.g. 'payment.approved_but_not_applied'
  title text not null,
  description text null,

  organization_id uuid null references public.organizations(id) on delete set null,
  user_id uuid null references public.users(id) on delete set null,

  provider text null,                    -- 'paypal'|'mercadopago'
  provider_payment_id text null,
  payment_id uuid null references public.payments(id) on delete set null,
  event_id uuid null references public.payment_events(id) on delete set null,

  fingerprint text null,                 -- para dedupe (hash estable)
  evidence jsonb not null default '{}'::jsonb,

  ack_by uuid null references public.users(id) on delete set null,
  ack_at timestamptz null,
  resolved_by uuid null references public.users(id) on delete set null,
  resolved_at timestamptz null
);

create index if not exists idx_ops_alerts_status on public.ops_alerts(status, severity, created_at desc);
create index if not exists idx_ops_alerts_org on public.ops_alerts(organization_id, created_at desc);
create unique index if not exists ux_ops_alerts_fingerprint_open
  on public.ops_alerts(fingerprint)
  where fingerprint is not null and status in ('open','ack');

-- updated_at trigger (si ya tenés update_updated_at_column(), usalo)
do $$
begin
  if exists (select 1 from pg_proc where proname = 'update_updated_at_column') then
    create trigger ops_alerts_set_updated_at
    before update on public.ops_alerts
    for each row
    execute function update_updated_at_column();
  end if;
end$$;


-- =========================================================
-- OPS: CHECK RUNS
-- =========================================================
create table if not exists public.ops_check_runs (
  id uuid primary key default gen_random_uuid(),
  created_at timestamptz not null default now(),

  check_suite text not null default 'ops_core',
  status text not null default 'success',   -- 'success'|'warning'|'error'
  duration_ms integer null,

  stats jsonb not null default '{}'::jsonb,  -- { alerts_opened: 2, alerts_resolved: 1, scanned: 120 }
  error_message text null
);

create index if not exists idx_ops_check_runs_created_at on public.ops_check_runs(created_at desc);


-- =========================================================
-- OPS: RUNBOOKS
-- =========================================================
create table if not exists public.ops_runbooks (
  id uuid primary key default gen_random_uuid(),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  alert_type text not null unique,
  title text not null,
  steps_md text not null,      -- markdown con pasos
  links jsonb not null default '[]'::jsonb
);

do $$
begin
  if exists (select 1 from pg_proc where proname = 'update_updated_at_column') then
    create trigger ops_runbooks_set_updated_at
    before update on public.ops_runbooks
    for each row
    execute function update_updated_at_column();
  end if;
end$$;

declare
  -- Usuario
  new_user_id uuid := gen_random_uuid();
  auth_user_id uuid := new.id;
  user_full_name text := coalesce(new.raw_user_meta_data->>'full_name', null);
  user_email text := new.email;
  user_avatar text := coalesce(new.raw_user_meta_data->>'avatar_url', '');
  avatar_source text := coalesce(new.raw_user_meta_data->>'avatar_source', '');
  provider text := coalesce(new.raw_user_meta_data->>'provider', 'email');

  -- Plan por defecto (free) → solo en organizaciones
  plan_free_id uuid := '015d8a97-6b6e-4aec-87df-5d1e6b0e4ed2';

  -- Rol global
  archub_user_role_id uuid := 'e6cc68d2-fc28-421b-8bd3-303326ef91b8';

  -- Organización
  org_id uuid := gen_random_uuid();
  org_name text := 'Organización de ' || coalesce(user_full_name, 'Usuario');

  -- Fecha
  now_timestamp timestamp with time zone := now();

  -- Rol organizacional
  org_member_id uuid := gen_random_uuid();
  admin_role_id uuid := '81be9c24-1e85-4f48-a131-2c618e0d1888';

  -- Moneda y billetera
  default_currency_id uuid := '58c50aa7-b8b1-4035-b509-58028dd0e33f';
  default_wallet_id uuid := '2658c575-0fa8-4cf6-85d7-6430ded7e188';
  currency_link_id uuid := gen_random_uuid();
  wallet_link_id uuid := gen_random_uuid();

  -- PDF
  default_pdf_template_id uuid := 'b6266a04-9b03-4f3a-af2d-f6ee6d0a948b';
begin
  RAISE LOG 'NEW USER START auth_id=% email=%', new.id, new.email;

  -- Paso 1: Crear usuario
  insert into public.users (
    id, auth_id, email, full_name, avatar_url, avatar_source, role_id
  ) values (
    new_user_id, auth_user_id, user_email, user_full_name, user_avatar, avatar_source, archub_user_role_id
  );
  RAISE LOG 'OK users';

  -- Paso 2: Crear user_data vacío
  insert into public.user_data (
    id, user_id, country, birthdate, created_at
  ) values (
    gen_random_uuid(), new_user_id, null, null, now_timestamp
  );
  RAISE LOG 'OK user_data';

  -- Paso 3: Crear organización con plan FREE
  insert into public.organizations (
    id, name, created_by, created_at, updated_at, is_active, plan_id
  ) values (
    org_id, org_name, new_user_id, now_timestamp, now_timestamp, true, plan_free_id
  );
  RAISE LOG 'OK organizations';

  -- Paso 4: Crear organization_data vacía
  insert into public.organization_data (
    organization_id
  ) values (
    org_id
  );
  RAISE LOG 'OK organization_data';

  -- Paso 5: Crear miembro admin
  insert into public.organization_members (
    id, user_id, organization_id, role_id, is_active, created_at, joined_at, last_active_at
  ) values (
    org_member_id, new_user_id, org_id, admin_role_id, true, now_timestamp, now_timestamp, now_timestamp
  );
  RAISE LOG 'OK organization_members';

  -- Paso 6: Moneda y billetera por defecto
  insert into public.organization_currencies (
    id, organization_id, currency_id, is_active, is_default, created_at
  ) values (
    currency_link_id, org_id, default_currency_id, true, true, now_timestamp
  );
  RAISE LOG 'OK organization_currencies';

  insert into public.organization_wallets (
    id, organization_id, wallet_id, is_active, is_default, created_at
  ) values (
    wallet_link_id, org_id, default_wallet_id, true, true, now_timestamp
  );
  RAISE LOG 'OK organization_wallets';

  -- Paso 7: Preferencias de la organización (incluye use_currency_exchange = false)
  insert into public.organization_preferences (
    organization_id,
    default_currency_id,
    default_wallet_id,
    default_pdf_template_id,
    use_currency_exchange,
    created_at,
    updated_at
  ) values (
    org_id,
    default_currency_id,
    default_wallet_id,
    default_pdf_template_id,
    false, -- 🔥 por defecto siempre en false
    now_timestamp,
    now_timestamp
  );
  RAISE LOG 'OK organization_preferences';

  -- Paso 8: Preferencias del usuario
  insert into public.user_preferences (
    id, user_id, last_organization_id, theme, onboarding_completed, sidebar_docked, created_at
  ) values (
    gen_random_uuid(), new_user_id, org_id, 'light', false, false, now_timestamp
  );
  RAISE LOG 'OK user_preferences';

  -- Paso 9: Crear user_organization_preferences
  insert into public.user_organization_preferences (
    id, user_id, organization_id, last_project_id, created_at, updated_at
  ) values (
    gen_random_uuid(), new_user_id, org_id, null, now_timestamp, now_timestamp
  );
  RAISE LOG 'OK user_organization_preferences';

  -- Paso 10: Vincular cuenta si no existe
  if not exists (
    select 1 from public.linked_accounts
    where auth_id = auth_user_id and provider_source = provider
  ) then
    insert into public.linked_accounts (
      id, user_id, auth_id, provider_source, created_at
    ) values (
      gen_random_uuid(), new_user_id, auth_user_id, provider, now_timestamp
    );
    RAISE LOG 'OK linked_accounts';
  end if;

  RAISE LOG 'NEW USER END id=% email=%', new_user_id, user_email;

  return new;

exception when others then
  raise exception '💥 ERROR real en archub_new_user: %', sqlerrm;
end;

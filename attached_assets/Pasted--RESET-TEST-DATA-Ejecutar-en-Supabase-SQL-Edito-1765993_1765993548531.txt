-- ============================================
-- RESET TEST DATA - Ejecutar en Supabase SQL Editor
-- ============================================

DO $$
DECLARE
    v_org_id UUID := '6d617475-c2f1-4e3d-b924-0c8dbdd44e25';  -- Tu organization_id
    v_user_id UUID := '0776911d-ccd9-4ac2-95c2-c1d7e270585b'; -- Tu user_id (opcional)
    v_free_plan_id UUID;
BEGIN
    -- Obtener el ID del plan Free
    SELECT id INTO v_free_plan_id FROM plans WHERE slug = 'free' LIMIT 1;
    
    IF v_free_plan_id IS NULL THEN
        RAISE EXCEPTION 'Plan Free no encontrado';
    END IF;
    
    -- 1. Eliminar organization_billing_cycles (depende de organization_subscriptions)
    DELETE FROM organization_billing_cycles WHERE subscription_id IN (
        SELECT id FROM organization_subscriptions WHERE organization_id = v_org_id
    );
    RAISE NOTICE 'Billing cycles eliminados';
    
    -- 2. Eliminar payment_events
    DELETE FROM payment_events 
    WHERE provider_payment_id IN (
        SELECT provider_payment_id FROM payments WHERE organization_id = v_org_id
    );
    RAISE NOTICE 'Payment events eliminados';
    
    -- 3. Eliminar payments
    DELETE FROM payments WHERE organization_id = v_org_id;
    RAISE NOTICE 'Payments eliminados';
    
    -- 4. Eliminar mp_subscription_preferences
    DELETE FROM mp_subscription_preferences WHERE organization_id = v_org_id;
    RAISE NOTICE 'MP Preferences eliminados';
    
    -- 5. Eliminar organization_subscriptions
    DELETE FROM organization_subscriptions WHERE organization_id = v_org_id;
    RAISE NOTICE 'Suscripciones eliminadas';
    
    -- 6. Eliminar course_lesson_progress del usuario
    IF v_user_id IS NOT NULL THEN
        DELETE FROM course_lesson_progress WHERE user_id = v_user_id;
        RAISE NOTICE 'Progreso de cursos eliminado';
    END IF;
    
    -- 7. Resetear organización al plan Free
    UPDATE organizations SET plan_id = v_free_plan_id WHERE id = v_org_id;
    RAISE NOTICE 'Organización reseteada a Free';
    
    RAISE NOTICE '=== RESET COMPLETADO ===';
END $$;

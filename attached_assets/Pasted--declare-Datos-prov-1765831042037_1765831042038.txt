
declare
  -- ============================================================
  --  🧑‍💻 Datos provenientes del auth.users (NEW)
  -- ============================================================
  new_user_id uuid := gen_random_uuid();
  auth_user_id uuid := new.id;
  user_full_name text := coalesce(new.raw_user_meta_data->>'full_name', null);
  user_email text := lower(new.email);
  user_avatar text := coalesce(new.raw_user_meta_data->>'avatar_url', '');
  provider text := lower(coalesce(new.raw_user_meta_data->>'provider', 'email'));

  avatar_src public.avatar_source_t := 'email'::public.avatar_source_t;

  -- ============================================================
  --  🎛 Valores globales
  -- ============================================================
  plan_free_id uuid := '015d8a97-6b6e-4aec-87df-5d1e6b0e4ed2';
  archub_user_role_id uuid := 'e6cc68d2-fc28-421b-8bd3-303326ef91b8';

  default_currency_id uuid := '58c50aa7-b8b1-4035-b509-58028dd0e33f';
  default_wallet_id uuid := '2658c575-0fa8-4cf6-85d7-6430ded7e188';
  default_pdf_template_id uuid := 'b6266a04-9b03-4f3a-af2d-f6ee6d0a948b';

  -- ============================================================
  --  🏢 Organización
  -- ============================================================
  org_id uuid := gen_random_uuid();
  org_name text := 'Organización de ' || coalesce(user_full_name, 'Usuario');

  -- ============================================================
  --  👥 Roles de la organización (🔥 CLAVE)
  -- ============================================================
  admin_role_id uuid := gen_random_uuid();
  editor_role_id uuid := gen_random_uuid();
  viewer_role_id uuid := gen_random_uuid();

  -- ============================================================
  --  ⏱️ Tiempo
  -- ============================================================
  now_timestamp timestamptz := now();

begin
  -- ============================================================
  --  Avatar source
  -- ============================================================
  if provider = 'google' then
    avatar_src := 'google';
  elsif provider = 'discord' then
    avatar_src := 'discord';
  else
    avatar_src := 'email';
  end if;

  -- ============================================================
  -- 1) users
  -- ============================================================
  insert into public.users (
    id, auth_id, email, full_name, avatar_url, avatar_source, role_id
  )
  values (
    new_user_id, auth_user_id, user_email, user_full_name, user_avatar, avatar_src, archub_user_role_id
  );

  -- ============================================================
  -- 2) user_data
  -- ============================================================
  insert into public.user_data (id, user_id, created_at)
  values (gen_random_uuid(), new_user_id, now_timestamp);

  -- ============================================================
  -- 3) organizations
  -- ============================================================
  insert into public.organizations (
    id, name, created_by, owner_id, created_at, updated_at, is_active, plan_id
  )
  values (
    org_id, org_name, new_user_id, new_user_id, now_timestamp, now_timestamp, true, plan_free_id
  );

  -- ============================================================
  -- 4) organization_data
  -- ============================================================
  insert into public.organization_data (organization_id)
  values (org_id);

  -- ============================================================
  -- 4.5) ROLES BASE DE LA ORGANIZACIÓN (CORREGIDO)
  -- ============================================================
  insert into public.roles (
    id, slug, name, description, type, organization_id, is_system, created_at
  ) values
    (admin_role_id,  'admin',  'Administrador', 'Acceso total', 'organization', org_id, false, now_timestamp),
    (editor_role_id, 'editor', 'Editor',        'Puede editar contenidos', 'organization', org_id, false, now_timestamp),
    (viewer_role_id, 'viewer', 'Lector',        'Solo lectura', 'organization', org_id, false, now_timestamp);

  -- ============================================================
  -- 5) organization_members (creator = admin)
  -- ============================================================
  insert into public.organization_members (
    id, user_id, organization_id, role_id, is_active, created_at, joined_at, last_active_at
  )
  values (
    gen_random_uuid(), new_user_id, org_id, admin_role_id,
    true, now_timestamp, now_timestamp, now_timestamp
  );

  -- (el resto de tu función queda IGUAL)

  return new;

exception
  when others then
    raise exception '💥 ERROR en handle_new_user: %', sqlerrm;
end;

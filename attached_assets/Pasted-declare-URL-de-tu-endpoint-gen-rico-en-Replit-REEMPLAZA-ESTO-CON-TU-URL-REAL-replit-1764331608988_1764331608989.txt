declare
  -- URL de tu endpoint genérico en Replit
  -- ¡REEMPLAZA ESTO CON TU URL REAL!
  replit_url text := 'https://seencel.com/api/email/send';

  -- Variables para el asunto y contenido del mail
  email_subject text;
  email_html text;
  user_email text;
begin
  -- --- LÓGICA 1: Nuevo Usuario Registrado ---
  if TG_TABLE_NAME = 'users' and TG_TABLE_SCHEMA = 'auth' and TG_OP = 'INSERT' then
    -- El email del usuario está en el nuevo registro (NEW.email)
    user_email := NEW.email;
    email_subject := '¡Bienvenido a Seencel! 🏗️';
    -- Podemos usar datos del registro para personalizar el HTML
    email_html := '<h1>Hola ' || coalesce(NEW.raw_user_meta_data->>'full_name', 'Arquitecto') || '!</h1><p>Gracias por unirte a nuestra plataforma de gestión de proyectos de arquitectura.</p><p>Estamos felices de tenerte.</p>';

  -- --- LÓGICA 2: Nueva Compra de Curso ---
  -- Asumo que tu tabla se llama 'purchases' en el esquema 'public' y tiene 'user_id' y 'course_name'
  elsif TG_TABLE_NAME = 'purchases' and TG_TABLE_SCHEMA = 'public' and TG_OP = 'INSERT' then
    -- Necesitamos buscar el email del usuario usando su ID
    select email into user_email from auth.users where id = NEW.user_id;

    email_subject := 'Confirmación de Compra: ' || NEW.course_name;
    email_html := '<h1>¡Felicitaciones por tu compra!</h1><p>Has adquirido el curso: <strong>' || NEW.course_name || '</strong>.</p><p>Monto pagado: $' || NEW.amount || '</p><p>Tu ID de transacción es: ' || NEW.id || '</p>';

  -- Si no es un evento que nos interese, no hacemos nada.
  else
    return NEW;
  end if;

  -- --- ENVIAR A REPLIT ---
  -- Si tenemos un email y asunto, hacemos la llamada HTTP
  if user_email is not null and email_subject is not null then
    perform
      extensions.http((
        'POST',
        replit_url,
        ARRAY[extensions.http_header('Content-Type', 'application/json')],
        'application/json',
        jsonb_build_object(
          'to', user_email,
          'subject', email_subject,
          'html', email_html
        )::text
      )::extensions.http_request);
  end if;

  return NEW;
end;
